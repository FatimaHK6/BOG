Usage in RequestActionBL

public async Task<CaseRegistrationRequestVM> RegisterCaseAsync(int requestId)
{
    var request = await _requestRepository.GetWithDetailsAsync(requestId);

    // Prepare case data
    var caseData = new CaseRegistrationData
    {
        RequestId = request.Id,
        CourtId = request.CourtId,
        DepartmentId = request.DepartmentId,
        DisputeTypeId = request.DisputeTypeId,
        Subject = request.Subject,
        Claims = request.Claims,
        Plaintiffs = request.Plaintiffs.Select(p => new CasePlaintiffData
        {
            FullName = p.FullName,
            IdentityNumber = p.IdentityNumber,
            // ...
        }).ToList(),
        Defendants = request.Defendants.Select(d => new CaseDefendantData
        {
            FullName = d.FullName,
            // ...
        }).ToList()
    };

    // Register case in system
    var result = await _caseManagementService.RegisterCaseAsync(caseData);

    if (!result.IsSuccess)
    {
        throw new BusinessException($"Case registration failed: {result.ErrorMessageAr}");
    }

    // Update request with case number
    request.CaseNumber = result.CaseNumber;
    request.RegistrationNumber = result.RegistrationNumber;
    request.RegistrationDate = result.RegistrationDate;
    request.RequestStatusId = (int)RequestStatus.Registered;

    await _unitOfWork.SaveChangesAsync();

    // Send notifications...
}

8. Test Scenarios

  1. Create New Request
    • POST /api/case-requests (create draft)
    • Verify status = Draft (1)
  2. Add Defendant
    • POST /api/case-requests/1/defendants
    • Verify only FullName is required
  3. Submit Request
    • POST /api/case-requests/1/submit
    • Verify all validation errors
  4. Take Action - Register
    • POST /api/case-requests/1/action with Action="Register"
    • Verify status changes to Registered (6)
    • Verify SMS and Email sent
  5. Take Action - Request Completion
    • POST /api/case-requests/1/action with Action="RequestCompletion"
    • Verify deficiencies added
    • Verify CompletionDeadline set
  6. Search Requests
    • GET /api/case-requests/search with various criteria
    • Verify pagination works
  7. Duplicate Check
    • Add same defendant twice
    • Verify ERR013 returned
  8. Case Management Integration
    • Register request
    • Verify case number received from case management system
