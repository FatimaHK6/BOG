3.3 Core Entity: CaseRegistrationRequest

public class CaseRegistrationRequest : BaseEntity
{
    public int RequestStatusId { get; set; }
    public string Subject { get; set; }                // 4000 chars
    public string Evidence { get; set; }               // 4000 chars
    public string Notes { get; set; }
    public int CourtId { get; set; }
    public DateTime? SubmissionDate { get; set; }
    public DateTime? CompletionDeadline { get; set; }  // 30 days
    public string RejectionReason { get; set; }

    public int CreatedByUserId { get; set; }
    public int? LastModifiedByUserId { get; set; }

    // Navigation
    public virtual RequestStatus Status { get; set; }
    public virtual Court Court { get; set; }
    public virtual User CreatedByUser { get; set; }

    // Many-to-Many
    public virtual ICollection<CaseRequestPlaintiff> CaseRequestPlaintiffs { get; set; }
    public virtual ICollection<CaseRequestDefendant> CaseRequestDefendants { get; set; }

    // One-to-Many
    public virtual ICollection<Claim> Claims { get; set; }
    public virtual ICollection<RelatedCase> RelatedCases { get; set; }
    public virtual ICollection<RequestClassification> Classifications { get; set; }
    public virtual ICollection<RequestAttachment> Attachments { get; set; }
}

7.5 Services

// plaintiff.service.ts
@Injectable({ providedIn: 'root' })
export class PlaintiffService {
  private baseUrl = '/api';

  constructor(private http: HttpClient) {}

  getPlaintiffs(requestId: number): Observable<Plaintiff[]> {
    return this.http.get<Plaintiff[]>(`${this.baseUrl}/case-requests/${requestId}/plaintiffs`);
  }

  getPlaintiff(id: number): Observable<Plaintiff> {
    return this.http.get<Plaintiff>(`${this.baseUrl}/plaintiffs/${id}`);
  }

  createPlaintiff(requestId: number, plaintiff: PlaintiffCreateDto): Observable<Plaintiff> {
    return this.http.post<Plaintiff>(`${this.baseUrl}/case-requests/${requestId}/plaintiffs`, plaintiff);
  }

  updatePlaintiff(id: number, plaintiff: PlaintiffUpdateDto): Observable<Plaintiff> {
    return this.http.put<Plaintiff>(`${this.baseUrl}/plaintiffs/${id}`, plaintiff);
  }

  deletePlaintiff(id: number): Observable<void> {
    return this.http.delete<void>(`${this.baseUrl}/plaintiffs/${id}`);
  }

  setAsApplicant(id: number): Observable<void> {
    return this.http.post<void>(`${this.baseUrl}/plaintiffs/${id}/set-applicant`, {});
  }
}

// absher.service.ts
@Injectable({ providedIn: 'root' })
export class AbsherService {
  private baseUrl = '/api/integration/absher';

  verifyIdentity(identityNumber: string, identityType: number): Observable<AbsherVerificationResult> {
    return this.http.post<AbsherVerificationResult>(`${this.baseUrl}/verify`, {
      identityNumber, identityType
    });
  }

  getPersonData(identityNumber: string, identityType: number): Observable<AbsherPersonData> {
    return this.http.get<AbsherPersonData>(`${this.baseUrl}/person/${identityNumber}?type=${identityType}`);
  }
}
