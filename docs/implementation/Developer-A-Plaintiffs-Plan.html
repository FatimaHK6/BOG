تكامل أبشر (Absher Integration)

يستخدم هذا الموديول خدمة أبشر للتحقق من هوية المواطنين والمقيمين وجلب بياناتهم الشخصية.

جدول مصدر البيانات (DataSource)

المعرفالاسمالاسم بالعربيةالوصف
1FromAbsherمن أبشرالبيانات مجلوبة من نظام أبشر
2FromUserمن المستخدمالبيانات مدخلة يدوياً

ملاحظة: استخدم DataSourceId بدلاً من IsAbsherVerified لتتبع مصدر بيانات المدعي أو الممثل.

الواجهة (Interface)

public interface IAbsherService
{
    Task<AbsherVerificationResult> VerifyIdentityAsync(
        string identityNumber,
        int identityType);

    Task<AbsherPersonData?> GetPersonDataAsync(
        string identityNumber,
        int identityType);

    Task<AbsherAddressData?> GetNationalAddressAsync(
        string identityNumber);
}

استخدام الخدمة في PlaintiffBL

public class PlaintiffBL : IPlaintiffBL
{
    private readonly IAbsherService _absherService;

    public async Task<PlaintiffVM> CreatePlaintiffAsync(
        PlaintiffCreateDTO dto)
    {
        // للأفراد (PlaintiffTypeId = 1 أو 2)
        if (dto.PlaintiffTypeId == 1 &&
            !string.IsNullOrEmpty(dto.IdentityNumber))
        {
            // التحقق من الهوية عبر أبشر
            var verification = await _absherService
                .VerifyIdentityAsync(dto.IdentityNumber, dto.IdentityTypeId);

            if (!verification.IsVerified)
            {
                throw new BusinessException(
                    "الهوية غير موجودة في أبشر");
            }

            // جلب البيانات الشخصية
            var personData = await _absherService
                .GetPersonDataAsync(dto.IdentityNumber, dto.IdentityTypeId);

            if (personData != null)
            {
                // ملء البيانات تلقائياً
                dto.FirstName = personData.FirstName;
                dto.FatherName = personData.FatherName;
                dto.GrandfatherName = personData.GrandfatherName;
                dto.FamilyName = personData.FamilyName;
                dto.BirthDate = personData.BirthDate;
                dto.Gender = personData.Gender;
            }
        }

        // إنشاء المدعي
        var plaintiff = _mapper.Map<Plaintiff>(dto);
        // DataSourceId: 1 = FromAbsher (من أبشر), 2 = FromUser (من المستخدم)
        plaintiff.DataSourceId = dto.PlaintiffTypeId == 1 ? 1 : 2;

        // متابعة إنشاء المدعي...
    }
}

ربط أنواع الممثلين بأنواع المدعين

كل نوع مدعي له أنواع ممثلين محددة مسموح بها:

نوع المدعيالممثلين المسموح بهم
فرد (Individual)محامي، وكيل، ولي، وصي
فرد بدون هويةمحامي، وكيل، ولي، وصي
صاحب مؤسسةمحامي، وكيل
شركة مسجلةمحامي، مصفي، أمين تفليسة، حارس قضائي، ممثل نظامي
شركة غير مسجلةمحامي، مصفي، أمين تفليسة، حارس قضائي، ممثل نظامي
جهة حكوميةممثل جهة حكومية
جمعية/مؤسسة أهليةمحامي، ممثل نظامي
وقفمحامي، ناظر

كود التحقق

private static readonly Dictionary<int, int[]> AllowedRepresentativeTypes = new()
{
    [1] = new[] { 1, 2, 3, 4 },       // فرد: محامي، وكيل، ولي، وصي
    [2] = new[] { 1, 2, 3, 4 },       // فرد بدون هوية
    [3] = new[] { 1, 2 },             // صاحب مؤسسة: محامي، وكيل
    [4] = new[] { 1, 5, 6, 7, 8 },    // شركة مسجلة
    [5] = new[] { 1, 5, 6, 7, 8 },    // شركة غير مسجلة
    [6] = new[] { 9 },               // جهة حكومية
    [7] = new[] { 1, 8 },            // جمعية/مؤسسة أهلية
    [8] = new[] { 1, 10 }            // وقف: محامي، ناظر
};

public void ValidateRepresentativeType(int plaintiffTypeId, int representativeTypeId)
{
    if (!AllowedRepresentativeTypes.TryGetValue(plaintiffTypeId, out var allowed))
        throw new BusinessException("نوع المدعي غير معروف");

    if (!allowed.Contains(representativeTypeId))
        throw new BusinessException("نوع الممثل غير مسموح لهذا النوع من المدعين");
}

سيناريوهات الاختبار

  • إضافة مدعي فرد (PlaintiffTypeId=1) مع جميع حقول البيانات الشخصية
  • إضافة مدعي شركة (PlaintiffTypeId=4) مع السجل التجاري
  • إضافة مدعي جهة حكومية (PlaintiffTypeId=6)
  • إضافة ممثل للمدعي (التحقق من صحة ربط الأنواع)
  • تعيين المدعي كمقدم طلب (التحقق أن الفرد فقط مسموح)
  • إضافة نفس المدعي مرتين (التحقق من ERR011)
  • إضافة ممثل بنفس رقم هوية المدعي (التحقق من ERR012)
  • رفع مرفق (التحقق PDF فقط، الحد الأقصى 4MB)
  • تعيين العنوان المختار عند اختلاف مدينة المدعي عن المحكمة
  • التحقق من تكامل أبشر للأفراد (جلب البيانات تلقائياً)
