التطبيق المطلوب:

Backend (FluentValidation)
// PlaintiffCreateDTOValidator.cs
RuleFor(x => x.NationalityId)
    .NotNull()
    .When(x => x.IdentityTypeId == 1)
    .WithMessage("يجب تعيين الجنسية السعودية");

RuleFor(x => x.NationalityId)
    .NotNull()
    .Must(id => id != SAUDI_NATIONALITY_ID)
    .When(x => x.IdentityTypeId == 2 || x.IdentityTypeId == 3)
    .WithMessage("لا يمكن اختيار الجنسية السعودية لهذا النوع من الهوية");
Frontend (TypeScript)
// plaintiff-form.component.ts
onIdentityTypeChange(typeId: number) {
  const nationalityControl = this.plaintiffForm.get('nationalityId');

  if (typeId === 1) { // هوية وطنية
    nationalityControl?.setValue(SAUDI_NATIONALITY_ID);
    nationalityControl?.disable();
  } else { // مقيم أو جواز
    nationalityControl?.enable();
    nationalityControl?.setValue(null);
    // فلترة قائمة الجنسيات لاستثناء السعودي
    this.filteredNationalities = this.nationalities
      .filter(n => n.id !== SAUDI_NATIONALITY_ID);
  }
}

❌ حقل مفقود: حالة العمل (employmentStatusId)

هذا الحقل غير موجود في Backend أو Frontend. يجب إضافته أولاً:

  • Entity: public int? EmploymentStatusId { get; set; }
  • Lookup Table: EmploymentStatus (1=حكومي, 2=خاص, 3=بدون عمل)
  • Frontend: dropdown + conditional logic

التطبيق المطلوب:

Backend (FluentValidation)
// PlaintiffCreateDTOValidator.cs
// BC01: جهة العمل مطلوبة إذا حكومي أو خاص
RuleFor(x => x.Employer)
    .NotEmpty()
    .When(x => x.EmploymentStatusId == 1 || x.EmploymentStatusId == 2)
    .WithMessage("جهة العمل مطلوبة");

RuleFor(x => x.Employer)
    .Null()
    .When(x => x.EmploymentStatusId == 3)
    .WithMessage("لا يمكن إدخال جهة العمل عند اختيار بدون عمل");

// BC02: عنوان العمل مطلوب إذا خاص فقط
RuleFor(x => x.WorkAddress)
    .NotNull()
    .When(x => x.EmploymentStatusId == 2)
    .WithMessage("عنوان العمل مطلوب للقطاع الخاص");

RuleFor(x => x.WorkAddress)
    .Null()
    .When(x => x.EmploymentStatusId == 1 || x.EmploymentStatusId == 3)
    .WithMessage("لا يمكن إدخال عنوان العمل");
Frontend (TypeScript)
// plaintiff-form.component.ts
onEmploymentStatusChange(statusId: number) {
  const employerControl = this.plaintiffForm.get('employer');
  const workAddressGroup = this.plaintiffForm.get('workAddress');

  // BC01: جهة العمل
  if (statusId === 1 || statusId === 2) { // حكومي أو خاص
    this.showEmployer = true;
    employerControl?.setValidators([Validators.required]);
  } else { // بدون عمل
    this.showEmployer = false;
    employerControl?.clearValidators();
    employerControl?.setValue(null);
  }

  // BC02: عنوان العمل
  if (statusId === 2) { // خاص فقط
    this.showWorkAddress = true;
  } else { // حكومي أو بدون عمل
    this.showWorkAddress = false;
    workAddressGroup?.reset();
  }

  employerControl?.updateValueAndValidity();
}

// plaintiff-form.component.ts
onWaqfOversightChange(type: string) {
  const agencyControl = this.plaintiffForm.get('waqfAgencyName');

  if (type === 'حكومية') {
    this.showWaqfAgencyName = true;
    agencyControl?.setValidators([Validators.required,
      Validators.maxLength(200)]);
  } else { // خاصة (أهلي)
    this.showWaqfAgencyName = false;
    agencyControl?.clearValidators();
    agencyControl?.setValue(null);
  }
  agencyControl?.updateValueAndValidity();
}
